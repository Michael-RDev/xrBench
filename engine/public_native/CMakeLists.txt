cmake_minimum_required(VERSION 3.24)
project(xrBenchNativeOperators LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(XRGPU_NATIVE_PACKAGE "" CACHE PATH "Verified immutable native operator package")
set(XRGPU_NATIVE_LIB_DIR "${XRGPU_NATIVE_PACKAGE}/lib/android-arm64" CACHE PATH "Native runtime libraries")
set(NLOHMANN_JSON_INCLUDE "" CACHE PATH "Pinned JSON headers")
option(XRBENCH_SOFTWARE_TEST "Explicit host-only numerical validation" OFF)
if(ANDROID AND XRBENCH_SOFTWARE_TEST)
  message(FATAL_ERROR "Software test runtime cannot ship in Android artifacts")
endif()
if(NOT ANDROID AND NOT XRBENCH_SOFTWARE_TEST)
  message(FATAL_ERROR "Production benchmarks target native Android ARM64")
endif()
foreach(name xrgpu_operators xrgpu_extended_operators)
  add_library(${name} SHARED IMPORTED)
  if(ANDROID)
    set(binary "${XRGPU_NATIVE_LIB_DIR}/lib${name}.so")
  else()
    find_library(binary_${name} NAMES ${name} PATHS "${XRGPU_NATIVE_LIB_DIR}" NO_DEFAULT_PATH REQUIRED)
    set(binary "${binary_${name}}")
  endif()
  set_target_properties(${name} PROPERTIES IMPORTED_LOCATION "${binary}")
endforeach()
add_library(xrbench_operator_core STATIC benchmark.cpp)
target_include_directories(xrbench_operator_core PUBLIC "${XRGPU_NATIVE_PACKAGE}/include" "${NLOHMANN_JSON_INCLUDE}"
  "${XRGPU_NATIVE_PACKAGE}/source/kernels/_shared")
target_link_libraries(xrbench_operator_core PUBLIC xrgpu_operators xrgpu_extended_operators)
if(ANDROID)
  add_library(xrbench_native_operators SHARED jni.cpp)
  target_link_libraries(xrbench_native_operators PRIVATE xrbench_operator_core android log)
  target_link_options(xrbench_native_operators PRIVATE "-Wl,-z,max-page-size=16384")
else()
  target_compile_definitions(xrbench_operator_core PRIVATE XRBENCH_SOFTWARE_TEST=1)
  add_executable(xrbench_native_bench main.cpp)
  target_link_libraries(xrbench_native_bench PRIVATE xrbench_operator_core)
endif()
